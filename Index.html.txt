    <script>
        document.getElementById('lottoDate').valueAsDate = new Date();

        let entries = JSON.parse(localStorage.getItem('lotto_entries') || '[]');
        let blocked = JSON.parse(localStorage.getItem('lotto_blocked') || '[]');

        function saveToStorage() {
            localStorage.setItem('lotto_entries', JSON.stringify(entries));
            localStorage.setItem('lotto_blocked', JSON.stringify(blocked));
            render();
        }

        function addEntry() {
            const name = document.getElementById('custName').value.trim();
            const rawNums = document.getElementById('lottoNumbers').value.trim();
            const top = Number(document.getElementById('amountTop').value) || 0;
            const bot = Number(document.getElementById('amountBot').value) || 0;
            const tod = Number(document.getElementById('amountTod').value) || 0;

            if (!name || !rawNums || (top === 0 && bot === 0 && tod === 0)) {
                alert('กรุณากรอกชื่อลูกค้า เลข และจำนวนเงินอย่างน้อย 1 ช่อง');
                return;
            }

            const nums = rawNums.split(/[\s,]+/).filter(n => n.length > 0);

            nums.forEach(num => {
                const isClosed = blocked.some(b => b.num === num && b.type === 'ปิดรับ');
                if (!isClosed) {
                    entries.push({
                        id: Date.now() + Math.random(),
                        type: document.getElementById('lottoType').value,
                        date: document.getElementById('lottoDate').value,
                        name: name,
                        num: num,
                        top: top,
                        bot: bot,
                        tod: tod
                    });
                }
            });

            document.getElementById('lottoNumbers').value = '';
            document.getElementById('amountTop').value = '';
            document.getElementById('amountBot').value = '';
            document.getElementById('amountTod').value = '';
            saveToStorage();
        }

        function deleteEntry(id) {
            entries = entries.filter(e => e.id !== id);
            saveToStorage();
        }

        function quickAction(action) {
            const input = document.getElementById('lottoNumbers');
            let val = input.value.trim();
            if(!val) return;
            let nums = val.split(/[\s,]+/).filter(n => n.length > 0);
            let result = [...nums];

            if(action === 'reverse') {
                nums.forEach(n => {
                    if(n.length === 2 && n[0] !== n[1]) {
                        result.push(n[1] + n[0]);
                    } else if(n.length === 3) {
                        result.push(n[1]+n[2]+n[0], n[2]+n[0]+n[1]);
                    }
                });
            } else if(action === 'frontRun') {
                nums.forEach(n => {
                    let digit = n.slice(-1);
                    for(let i=0; i<=9; i++) result.push(i + digit);
                });
            } else if(action === 'backRun') {
                nums.forEach(n => {
                    let digit = n[0];
                    for(let i=0; i<=9; i++) result.push(digit + i);
                });
            }
            input.value = [...new Set(result)].join(', ');
        }

        function addBlockedNumber() {
            const num = document.getElementById('blockedNum').value.trim();
            const type = document.getElementById('blockedType').value;
            if(num) {
                blocked.push({num, type});
                document.getElementById('blockedNum').value = '';
                saveToStorage();
            }
        }

        function removeBlocked(index) {
            blocked.splice(index, 1);
            saveToStorage();
        }

        function exportData() {
            const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify({entries, blocked}));
            const dlAnchorElem = document.createElement('a');
            dlAnchorElem.setAttribute("href", dataStr);
            dlAnchorElem.setAttribute("download", "lotto_backup.json");
            dlAnchorElem.click();
        }

        function importData(event) {
            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const data = JSON.parse(e.target.result);
                    entries = data.entries || [];
                    blocked = data.blocked || [];
                    saveToStorage();
                    alert('นำเข้าข้อมูลสำเร็จ!');
                } catch(err) {
                    alert('ไฟล์ไม่ถูกต้อง');
                }
            };
            reader.readAsText(event.target.files[0]);
        }

        function render() {
            const tbody = document.getElementById('entryTableBody');
            tbody.innerHTML = '';
            
            let totalT = 0, totalB = 0, totalTo = 0;
            const summaryMap = {};
            const customerSummaryMap = {};
            const customers = new Set();

            entries.forEach(e => {
                customers.add(e.name);
                totalT += Number(e.top);
                totalB += Number(e.bot);
                totalTo += Number(e.tod);

                // 1. รวมยอดตามเลข
                ['top', 'bot', 'tod'].forEach(k => {
                    if(e[k] > 0) {
                        const key = `${e.num} (${k.toUpperCase()})`;
                        summaryMap[key] = (summaryMap[key] || 0) + Number(e[k]);
                    }
                });

                // 2. แยกยอดตามลูกค้า (แต่ละเจ้า)
                if (!customerSummaryMap[e.name]) {
                    customerSummaryMap[e.name] = { top: 0, bot: 0, tod: 0, total: 0 };
                }
                customerSummaryMap[e.name].top += Number(e.top);
                customerSummaryMap[e.name].bot += Number(e.bot);
                customerSummaryMap[e.name].tod += Number(e.tod);
                customerSummaryMap[e.name].total += Number(e.top) + Number(e.bot) + Number(e.tod);

                const tr = document.createElement('tr');
                tr.className = "hover:bg-slate-700/30";
                tr.innerHTML = `
                    <td class="p-2 font-medium">${e.name}</td>
                    <td class="p-2 font-mono text-amber-300 font-bold">${e.num}</td>
                    <td class="p-2 text-right">${e.top > 0 ? e.top : '-'}</td>
                    <td class="p-2 text-right">${e.bot > 0 ? e.bot : '-'}</td>
                    <td class="p-2 text-right">${e.tod > 0 ? e.tod : '-'}</td>
                    <td class="p-2 text-center">
                        <button onclick="deleteEntry(${e.id})" class="text-rose-400 hover:text-rose-300 p-1"><i class="fa-solid fa-trash"></i></button>
                    </td>
                `;
                tbody.appendChild(tr);
            });

            document.getElementById('totalEntriesCount').innerText = `${entries.length} รายการ`;

            // Datalist customers
            const datalist = document.getElementById('customerList');
            datalist.innerHTML = '';
            customers.forEach(c => {
                const opt = document.createElement('option');
                opt.value = c;
                datalist.appendChild(opt);
            });

            // Render: Summary by Customer (แยกยอดตามเจ้า/ลูกค้า)
            const custSumContainer = document.getElementById('customerSummary');
            custSumContainer.innerHTML = '';
            const sortedCust = Object.entries(customerSummaryMap).sort((a,b) => b[1].total - a[1].total);
            if(sortedCust.length === 0) {
                custSumContainer.innerHTML = '<p class="text-xs text-slate-500">ยังไม่มีข้อมูลยอดซื้อของลูกค้า</p>';
            } else {
                sortedCust.forEach(([name, data]) => {
                    custSumContainer.innerHTML += `
                        <div class="bg-slate-900 p-2.5 rounded border border-slate-700/50 space-y-1">
                            <div class="flex justify-between font-bold text-amber-300 text-sm">
                                <span><i class="fa-solid fa-user text-xs mr-1"></i> ${name}</span>
                                <span>ยอดรวม: ${data.total.toLocaleString()} บาท</span>
                            </div>
                            <div class="text-xs text-slate-400 flex justify-between">
                                <span>บน: ${data.top} | ล่าง: ${data.bot} | โต๊ด: ${data.tod}</span>
                            </div>
                        </div>`;
                });
            }

            // Render: Summary by Number (รวมยอดตามตัวเลข)
            const sumContainer = document.getElementById('numberSummary');
            sumContainer.innerHTML = '';
            const sortedSum = Object.entries(summaryMap).sort((a,b) => b[1] - a[1]);
            if(sortedSum.length === 0) {
                sumContainer.innerHTML = '<p class="text-xs text-slate-500">ยังไม่มีข้อมูลยอดซื้อ</p>';
            } else {
                sortedSum.forEach(([k, v]) => {
                    sumContainer.innerHTML += `<div class="flex justify-between bg-slate-900 p-2 rounded border border-slate-700/50"><span class="font-mono font-bold text-amber-300">${k}</span><span class="font-semibold">${v} บาท</span></div>`;
                });
            }

            // Blocked list
            const blockContainer = document.getElementById('blockedList');
            blockContainer.innerHTML = '';
            blocked.forEach((b, idx) => {
                blockContainer.innerHTML += `<span class="bg-rose-950 text-rose-300 border border-rose-800 text-xs px-2.5 py-1 rounded-full flex items-center gap-1">เลข ${b.num} (${b.type}) <button onclick="removeBlocked(${idx})" class="hover:text-white">&times;</button></span>`;
            });
        }

        render();
    </script>
